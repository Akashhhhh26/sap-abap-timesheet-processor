REPORT ZAK_PROJECT1.

TYPES : BEGIN OF TY_TIMESHEET,
          EMPID    TYPE CHAR10,
          PDATE    TYPE DATS,
          CLOCKIN  TYPE CHAR5,
          CLOCKOUT TYPE CHAR5,
          DEPTCODE TYPE CHAR4,
          STATUS   TYPE CHAR40,
        END OF TY_TIMESHEET.

DATA : GT_RAW        TYPE STANDARD TABLE OF TY_TIMESHEET,
       GT_VALID      TYPE STANDARD TABLE OF TY_TIMESHEET,
       GT_EXCEPTIONS TYPE STANDARD TABLE OF TY_TIMESHEET,
       LS_TS         TYPE TY_TIMESHEET.

DATA : LT_RAW_LINES TYPE TABLE OF STRING,
       LV_LINE      TYPE STRING,
       LV_HOURS     TYPE P DECIMALS 2.

CALL FUNCTION 'GUI_UPLOAD'
  EXPORTING
    FILENAME = 'C:\Users\akash\Downloads\timesheet.txt'
  TABLES
    DATA_TAB = LT_RAW_LINES.

LOOP AT LT_RAW_LINES INTO LV_LINE.
  SPLIT LV_LINE AT ','
  INTO LS_TS-EMPID LS_TS-PDATE LS_TS-CLOCKIN LS_TS-CLOCKOUT LS_TS-DEPTCODE.
  APPEND LS_TS TO GT_RAW.
ENDLOOP.

LOOP AT GT_RAW INTO LS_TS.

  IF LS_TS-CLOCKOUT < LS_TS-CLOCKIN.
    LS_TS-STATUS = 'INVALID - CLOCKOUT BEFORE CLOCKIN'.
    APPEND LS_TS TO GT_EXCEPTIONS.
  ELSE.
    LV_HOURS = ( LS_TS-CLOCKOUT - LS_TS-CLOCKIN ) / 100.
    IF LV_HOURS > 12.
      LS_TS-STATUS = 'FLAGGED - OVER 12 HOURS'.
      APPEND LS_TS TO GT_EXCEPTIONS.
    ELSE.
      LS_TS-STATUS = 'VALID'.
      APPEND LS_TS TO GT_VALID.
    ENDIF.
  ENDIF.

ENDLOOP.

WRITE: / 'TIMESHEET REPORT'.
WRITE:/ '==================='.
SKIP.

WRITE: / 'VALID RECORDS'.
LOOP AT GT_VALID INTO LS_TS.
  WRITE: / LS_TS-EMPID, LS_TS-PDATE, LS_TS-CLOCKIN, LS_TS-CLOCKOUT, LS_TS-DEPTCODE.
ENDLOOP.

SKIP 2.
WRITE: / 'EXCEPTIONS'.
LOOP AT GT_EXCEPTIONS INTO LS_TS.
  WRITE: / LS_TS-EMPID, LS_TS-PDATE, LS_TS-CLOCKIN, LS_TS-CLOCKOUT, LS_TS-DEPTCODE.
ENDLOOP.
